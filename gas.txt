const SHEET_NAME_QUESTIONS = "Questions";
const SHEET_NAME_ANSWERS = "Answers";
const SHEET_NAME_REQUIREMENT = "Requirement";
const FOLDER_ID = "1Cj9k_6BjWC8QFK7uXCt5ExeeDL2WhjTI";

// 1. Trả về danh sách câu hỏi khi màn hình HTML load
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_QUESTIONS);
  const data = sheet.getRange(1, 1, sheet.getLastRow(), 1).getValues();
  const questions = data.map(row => row[0]).filter(q => q !== "");

  const requirementSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_REQUIREMENT);
  const requirements = requirementSheet && requirementSheet.getLastRow() > 1
    ? requirementSheet.getRange(2, 1, requirementSheet.getLastRow() - 1, 1).getValues()
        .map(row => String(row[0] || "").trim()).filter(Boolean)
    : [];

  return ContentService.createTextOutput(JSON.stringify({ questions, requirements }))
    .setMimeType(ContentService.MimeType.JSON);
}

// 2. Nhận dữ liệu Text và Audio File từ HTML gửi lên
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const textAnswer = data.transcript;
    const question = data.question;
    const score = data.score;
    const matchedRequirements = data.matchedRequirements || [];
    const totalRequirements = data.totalRequirements || 0;
    const audioBase64 = data.audioBase64; // File âm thanh đã mã hóa
    
    let audioUrl = "";
    
    // Nếu có file âm thanh, giải mã và lưu vào Drive
    if (audioBase64) {
      const folder = DriveApp.getFolderById(FOLDER_ID);
      const decodedAudio = Utilities.base64Decode(audioBase64.split(',')[1]); 
      const blob = Utilities.newBlob(decodedAudio, 'audio/webm', `Record_${new Date().getTime()}.webm`);
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      audioUrl = file.getUrl();
    }
    
    // Ghi vào Sheet Answers
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_ANSWERS);
    // Existing columns A:D are preserved; score details are added in E:G.
    sheet.appendRow([new Date(), question, textAnswer, audioUrl, score, matchedRequirements.join(", "), totalRequirements]);
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.message})).setMimeType(ContentService.MimeType.JSON);
  }
}
